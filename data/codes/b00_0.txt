<script>
    (() => {
        const NBSP = '\u00A0';
        const MAX_LEN = 3;

        const W = '[\\p{L}\\p{N}]';
        const SYMBOLS = '[₽$€£¥%‰°]';

        const rules = [
            [new RegExp(`(?<!${W})(${W}{1,${MAX_LEN}})\\s+(?=\\S)`, 'gu'), `$1${NBSP}`],
            [new RegExp(`(\\d)\\s+(${SYMBOLS})`, 'gu'), `$1${NBSP}$2`],
            [new RegExp(`(\\d)\\s+(\\p{L}{1,${MAX_LEN}})(?!\\p{L})`, 'gu'), `$1${NBSP}$2`],
            [new RegExp(`\\s+(${W}{1,${MAX_LEN}}|${SYMBOLS})(?=[.!?…]+(?:\\s|$)|$)`, 'gu'), `${NBSP}$1`],
        ];

        const SKIP = new Set(['SCRIPT', 'STYLE', 'TEXTAREA', 'INPUT', 'CODE', 'PRE', 'NOSCRIPT', 'SVG']);

        const fix = text => rules.reduce((t, [re, rep]) => t.replace(re, rep), text);

        let fixedCount = 0;

        function fixTextNode(n) {
            const p = n.parentElement;
            if (!p || SKIP.has(p.tagName.toUpperCase()) || p.isContentEditable) return;
            const fixed = fix(n.nodeValue);
            if (fixed !== n.nodeValue) {
                n.nodeValue = fixed;
                fixedCount++;
            }
        }

        function processTree(root) {
            const walker = document.createTreeWalker(root, NodeFilter.SHOW_TEXT | NodeFilter.SHOW_ELEMENT);
            const textNodes = [];
            const shadowHosts = [];
            let node = root.nodeType === Node.TEXT_NODE ? root : walker.currentNode;

            if (root.nodeType === Node.TEXT_NODE) {
                fixTextNode(root);
                return;
            }

            do {
                if (node.nodeType === Node.TEXT_NODE) textNodes.push(node);
                else if (node.shadowRoot) shadowHosts.push(node.shadowRoot);
            } while ((node = walker.nextNode()));

            textNodes.forEach(fixTextNode);

            for (const sr of shadowHosts) {
                processTree(sr);
                observe(sr);
            }
        }

        // Очередь: собираем изменения и обрабатываем раз в кадр
        const queue = new Set();
        let scheduled = false;

        function schedule(target) {
            queue.add(target);
            if (scheduled) return;
            scheduled = true;
            requestAnimationFrame(() => {
                scheduled = false;
                const items = [...queue];
                queue.clear();
                items.forEach(t => t.isConnected && processTree(t));
            });
        }

        const observed = new WeakSet();

        function observe(target) {
            if (observed.has(target)) return;
            observed.add(target);
            new MutationObserver(mutations => {
                for (const m of mutations) {
                    if (m.type === 'characterData') schedule(m.target);
                    else m.addedNodes.forEach(n => schedule(n));
                }
            }).observe(target, {
                childList: true,
                subtree: true,
                characterData: true
            });
        }

        function start() {
            processTree(document.body);
            observe(document.body);
            console.log(`[nbsp] исправлено текстовых узлов: ${fixedCount}`);
        }

        if (document.readyState === 'complete') setTimeout(start, 500);
        else window.addEventListener('load', () => setTimeout(start, 500));
    })();
</script>
